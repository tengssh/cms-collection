optimizer = BayesianOptimizer(
    estimator=regressor,
    X_training=x_init,
    y_training=y_init,
    query_strategy=max_EI

#+RESULTS:

***** Run BO loop
- =query=: Selects the next x from the provided pool X using acquisition (maximum expected improvement, which balances exploration and exploitation)
- =teach=: Evaluates true function at x and retrain the surrogate (Gaussian process regressor)
- =predict=: Calculates posterior mean/std over all X
- =optimizer_EI=: Calculates acquisition function
- =get_max=: Returns the maximum estimate from the surrogate mean (pure exploitation)
- Iterate the process

#+begin_src python :results output :session *python* :exports both
class BayesOpt:
    def __init__(self, X: np.ndarray, y: np.ndarray, optimizer: "BayesianOptimizer"):
        self.X = X
        self.y = y
        self.optimizer = optimizer
        self.i_opt = 0

        self.query_idx = None
        self.query_inst = None
        self.X_max, self.y_max = self.optimizer.get_max()
        self.y_pred, self.y_std = self.optimizer.predict(self.X, return_std=True)
        self.y_pred, self.y_std = self.y_pred.ravel(), self.y_std.ravel()
        self.EI = optimizer_EI(self.optimizer, self.X)

    def step(self) -> "BayesOpt":
        # query
        self.query_idx, self.query_inst = self.optimizer.query(self.X)
        x_new = self.X[self.query_idx].reshape(-1, 1)
        y_new = self.y[self.query_idx].reshape(-1, 1)

        # teach
        self.optimizer.teach(x_new, y_new)

        # predict
        self.y_pred, self.y_std = self.optimizer.predict(self.X, return_std=True)
        self.y_pred, self.y_std = self.y_pred.ravel(), self.y_std.ravel()

        # acquisition
        self.EI = optimizer_EI(self.optimizer, self.X)

        # get_max
        self.X_max, self.y_max = self.optimizer.get_max()

        self.i_opt += 1
        return self

    def plot(self, fig_out: str | None = None):
        import matplotlib.pyplot as plt

        if fig_out is None:
            fig_out = f"bo_{self.i_opt}.png"

        with plt.style.context("seaborn-v0_8-white"):
            fig, (ax1, ax2) = plt.subplots(2, 1, figsize=(6, 6), sharex=True,
                gridspec_kw={"height_ratios": [2, 1], "hspace": 0.0},
            )

            # Surrogate
            ax1.scatter(self.optimizer.X_training, self.optimizer.y_training, c="k", s=50, label="Queried")
            ax1.scatter(self.X_max, self.y_max, s=100, c="r", label="Current")
            ax1.plot(self.X.ravel(), self.y_pred, label="GP")
            ax1.fill_between(self.X.ravel(), self.y_pred - self.y_std, self.y_pred + self.y_std, alpha=0.5)
            # True function
            ax1.plot(self.X.ravel(), self.y, c="k", linewidth=2, label="Truth")
            ax1.set_ylabel(r"-Energy (eV)")
            ax1.legend(loc="upper left")

            # Acquisition function
            ax2.plot(self.X.ravel(), self.EI, c="b", label="EI")
            ax2.axvline(self.X_max[0], color="r", linestyle="-", label="Current max")
            ax2.axvline(self.X[np.argmax(self.EI)], color="k", linestyle="--", label="Next query")
            ax2.set_xlabel(r"a ($\AA$)")
            ax2.set_ylabel("Expected Improvement")
            ax2.legend(loc="upper center", bbox_to_anchor=(0.5, 1.2), ncol=3)

        print("#+ATTR_org: :width 600")
        print(f"[[./{fig_out}]]")
        plt.savefig(fig_out, dpi=150, bbox_inches="tight")
        plt.close("all")
#+end_src

BO = BayesOpt(X, y, optimizer)
y_max_0 = BO.y_max
BO.plot()
for i in range(10):
    print(f">> Iter {BO.i_opt}")
    BO = BO.step()
    BO.plot()
    if np.abs(BO.y_max - y_max_0) <= 0.01 and np.max(BO.y_std) <= 0.035:
        print("Stop criterion met!")
        break
    if BO.y_max > y_max_0:
        y_max_0 = BO.y_max
