import os
import subprocess
import time
import numpy as np

sd_binary = os.environ.get("SD_BINARY")
commands = sd_binary.split()

INPSD = lambda simid = 'bccFe', L = 10, temp = 100: \
"""simid {SIMID}
ncell     {L}       {L}       {L}               System size            
BC        P         P         P                 Boundary conditions (0=vacuum, P=periodic)
cell      1.00000   0.00000   0.00000         
          0.00000   1.00000   0.00000
          0.00000   0.00000   1.00000
Sym       1                                     Symmetry of lattice (0 for no, 1 for cubic, 2 for 2d cubic, 3 for hexagonal)

posfile   ./posfile
momfile   ./momfile
exchange  ./jfile

SDEalgh   1                                     SDE solver: 1=midpoint, 2=heun, 3=heun3, 4=Heun_proper, 5=Depondt
Initmag   1                                     Initial config of moments (1=random, 2=cone, 3=spec., 4=file)
restartfile ./restart.{SIMID}.out
Mensemble 5

ip_mode   M                                     Initial phase parameters
ip_temp   {TEMP}                                --
ip_mcNstep     10000                            --

mode      S                                     S=SD, M=MC
temp      {TEMP}                                Measurement phase parameters
Nstep   20000 
damping 5.0
timestep 1.0e-16

do_avrg   Y                                     Measure averages

do_cumu Y
cumu_step 50
cumu_buff 10

do_tottraj N                                    Measure moments
tottraj_step   1000

plotenergy 1
""".format(SIMID=simid, L=L, TEMP=temp)

POSFILE = \
"""1 1   0.000000  0.000000  0.000000
2 1   0.500000  0.500000  0.500000
"""

MOMFILE = \
"""1 1 2.23 1.0 0.0 0.0 2.0
2 1 2.23 1.0 0.0 0.0 2.0
"""

JFILE = \
"""1 1  0.5       0.5       0.5       1.33767484769984
1 1  1.0       0.0       0.0       0.75703576545650
1 1  1.0       1.0       0.0       -0.05975437643846
1 1  1.5       0.5       0.5       -0.08819834160658
"""

def run_Tsweep(L = 10, SIMID = "bccFe100",
               T_list = None, f_out = "sum.txt", if_append = False):
    if not if_append:
        with open(f_out, 'w') as fw:
            fw.write("#Temp Iter <M> <M^2> <M^4> U_{Binder} \\chi C_v(tot) <E> <E_{exc}> <E_{lsf}>\n")

    if T_list is None:
        T_list = [100*_ for _ in range(17)]

    for temp in T_list:
        with open('inpsd.dat', 'w') as fw:
            fw.write(INPSD(simid=SIMID, L=L, temp=temp))
        with open('posfile', 'w') as fw:
            fw.write(POSFILE)
        with open('momfile', 'w') as fw:
            fw.write(MOMFILE)
        with open('jfile', 'w') as fw:
            fw.write(JFILE)
    
        print(f"# L = {L}, T = {temp} K")
        time_start = time.time()
        with open("out.log", "w") as fw:
            subprocess.run(commands,
                           stdout=fw,
                           stderr=subprocess.STDOUT,
                           text=True,)
        time_end = time.time()
        print(f">> Runtime: {time_end-time_start:.2f} s")
    
        with open(f"cumulants.{SIMID}.out", "r") as fr:
            lines = fr.readlines()
            last_line = lines[-1].strip()
        with open(f_out, 'a') as fa:
            fa.write(f"{temp} {last_line}\n")

# main program
for L in [5, 10, 15]:
    os.system("rm -f restart.*.out")

              900, 950, 975, 1000, 1025,
              1050, 1100, 1200, 1400, 1600]
    f_out = f"sum_{SIMID}.txt"
    run_Tsweep(L=L, SIMID=SIMID, T_list=T_list, f_out=f_out)
    run_Tsweep(L=L, SIMID=SIMID, T_list=T_list[::-1], f_out=f_out, if_append=True)

def load_sum(f_sum = "sum.txt"):
    data = np.loadtxt(f_sum)
    T_all = data[:, 0]  # Temperature
    M_all = data[:, 2]  # Magnetization
    U_all = data[:, 5]  # Binder cumulant
    S_all = data[:, 6]  # Susceptibility
    cv_all = data[:, 7] # Specific heat
    return T_all, M_all, U_all, S_all, cv_all

def plot_sum(f_sum = "sum.txt"):
    SIMID = f_sum.split('.')[0][4:]
    T_all, M_all, U_all, S_all, cv_all = load_sum(f_sum)
    
    fig, axes = plt.subplots(4, 1, figsize=(6, 8), sharex=True)
    ax_mag, ax_susc, ax_cv, ax_u = axes
    
    # --- Heating data (Red) ---
    line_sty = "-o"
    color = "r"
    label = "heating"
    
    idx0, idx1 = 0, len(T_all) // 2
    x = T_all[idx0:idx1]         # Temperature
    mag = M_all[idx0:idx1]       # Magnetization
    u_binder = U_all[idx0:idx1]  # Binder cumulant
    susc = S_all[idx0:idx1]      # Susceptibility
    cv = cv_all[idx0:idx1]       # Specific heat
    
    ax_mag.plot(x, mag / mag[0], line_sty, color=color, label=label)
    ax_susc.plot(x, susc / np.max(susc), line_sty, color=color, label=label)
    ax_cv.plot(x, cv / np.max(cv), line_sty, color=color, label=label)
    ax_u.plot(x, u_binder, line_sty, color=color, label=label)
    
    # --- Cooling data (Blue) ---
    line_sty = "--x"
    color = "b"
    label = "cooling"
    
    idx0, idx1 = len(T_all) // 2, len(T_all)
    x = T_all[idx0:idx1]         # Temperature
    mag = M_all[idx0:idx1]       # Magnetization
    u_binder = U_all[idx0:idx1]  # Binder cumulant
    susc = S_all[idx0:idx1]      # Susceptibility
    cv = cv_all[idx0:idx1]       # Specific heat
    
    ax_mag.plot(x, mag / mag[-1], line_sty, color=color, label=label)
    ax_susc.plot(x, susc / np.max(susc), line_sty, color=color, label=label)
    ax_cv.plot(x, cv / np.max(cv), line_sty, color=color, label=label)
    ax_u.plot(x, u_binder, line_sty, color=color, label=label)
    
    # --- Axis labels & Titles ---
    fontsize = "x-large"
    ax_mag.set_ylabel(r"$M / M_{0}$", fontsize=fontsize)
    ax_susc.set_ylabel(r"$\chi / \chi_{max}$", fontsize=fontsize)
    ax_cv.set_ylabel(r"$C_v / C_{v,max}$", fontsize=fontsize)
    ax_u.set_ylabel(r"$U$ (Binder)", fontsize=fontsize)
    ax_u.set_xlabel("T (K)", fontsize=fontsize)
    ax_mag.set_title(f"{SIMID}", fontsize=fontsize)
    
    # --- Styling ---
    for ax in axes:
        ax.spines["top"].set_visible(False)
        ax.spines["right"].set_visible(False)
        ax.grid(True, linestyle="--", alpha=0.6)
        ax.legend(frameon=False, loc="best", fontsize=fontsize)
    
    ax_mag.set_ylim(-0.05, 1.1)
    ax_susc.set_ylim(-0.05, 1.1)
    ax_cv.set_ylim(-0.05, 1.1)
    ax_u.set_xlim(0, 1600)
    
    fig_out = f"MT_{SIMID}.png"
    fig.tight_layout()
    plt.savefig(fig_out, bbox_inches='tight', dpi=150)
    print(f"#+ATTR_ORG: :width 600")
    print(f"[[file:{PATH+'/'+fig_out}]]")
    plt.savefig(fig_out, dpi=300, bbox_inches="tight")

        T_all, M_all, U_all, S_all, cv_all = load_sum(f_sum)
        Data[SIMID]['T'] = T_all
        Data[SIMID]['M'] = M_all

        mag = M_all[idx0:idx1]        # Magnetization

        susc = S_all[idx0:idx1]       # Susceptibility
        cv = cv_all[idx0:idx1]        # Specific heat
        
        line_sty = "o-"
        label = f"L={SIMID[6:]}"
        ax_mag.plot(x, mag / mag[0], line_sty, label=label)
        ax_susc.plot(x, susc / np.max(susc), line_sty, label=label)
        ax_cv.plot(x, cv / np.max(cv), line_sty, label=label)
        ax_u.plot(x, u_binder, line_sty, label=label)
